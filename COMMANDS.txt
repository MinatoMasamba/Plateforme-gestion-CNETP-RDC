📦 SETUP & INSTALLATION
───────────────────────────────────────────────────────────────────────────

# Activer l'environnement virtuel
source mon_env/bin/activate          # Linux/Mac
mon_env\Scripts\activate              # Windows

# Installer/mettre à jour les dépendances
pip install -r requirements.txt
pip install --upgrade pip

# Créer les migrations de la base de données
python manage.py makemigrations
python manage.py makemigrations apps.core
python manage.py makemigrations apps.experts
python manage.py makemigrations apps.governance
python manage.py makemigrations apps.norms

# Appliquer les migrations
python manage.py migrate

# Créer un superutilisateur (admin)
python manage.py createsuperuser

# Charger des données initiales
python manage.py loaddata data.json    # Si vous avez un fichier de données


🚀 DÉMARRAGE DU SERVEUR
───────────────────────────────────────────────────────────────────────────

# Serveur de développement simple
python manage.py runserver

# Serveur avec adresse personnalisée
python manage.py runserver 0.0.0.0:8000

# Serveur avec rechargement automatique
python manage.py runserver --nothreading

# Avec Gunicorn (production-like)
gunicorn config.wsgi:application --bind 0.0.0.0:8000 --workers 4


🐳 DOCKER & CONTENEURS
───────────────────────────────────────────────────────────────────────────

# Démarrer tous les services

docker-compose down

# Reconstruire les images
docker-compose build --no-cache
docker-compose up -d --force-recreate

# Vérifier l'état des services
docker-compose ps


🧪 TESTS & QUALITÉ DE CODE
───────────────────────────────────────────────────────────────────────────

# Exécuter les tests pytest
pytest tests/
pytest tests/test_models.py -v
pytest tests/ -k "test_user" -v
pytest tests/ --cov=apps --cov-report=html

# Exécuter les tests Django standard
python manage.py test apps.core
python manage.py test

# Tests avec verbosité
pytest tests/ -vv
pytest tests/ --tb=short
pytest tests/ --pdb              # Debugger au premier échec

flake8 .                          # Vérifier la qualité
isort .                           # Trier les imports

# Combiner linters
black . && flake8 . && isort .

🎯 WORKFLOW DÉVELOPPEMENT TYPIQUE
───────────────────────────────────────────────────────────────────────────

1. Activer l'environnement
   source mon_env/bin/activate

2. Créer une branche feature
   git checkout -b feature/new-feature

3. Faire les changements dans les modèles
   vim apps/experts/models.py

4. Créer les migrations
   python manage.py makemigrations

5. Tester les migrations
   python manage.py migrate

6. Créer serializers et views
   vim apps/experts/serializers.py
   vim apps/experts/views.py

7. Lancer les tests
   pytest tests/

8. Formater le code
   black . && flake8 . && isort .

9. Commiter et pusher
   git add .
   git commit -m "Add new feature"
   git push origin feature/new-feature


🔐 VARIABLES D'ENVIRONNEMENT (.env)
───────────────────────────────────────────────────────────────────────────

Créer un fichier .env à la racine avec :

SECRET_KEY=your-secret-key-here
DEBUG=False
ALLOWED_HOSTS=localhost,127.0.0.1

# PostgreSQL
DB_NAME=cnetp_db
DB_USER=cnetp_user
DB_PASSWORD=secure_password

DB_PORT=5432

# Redis

# Celery
CELERY_BROKER_URL=redis://localhost:6379/0
CELERY_RESULT_BACKEND=redis://localhost:6379/0

# Email

EMAIL_USE_TLS=True
EMAIL_HOST_USER=your-email@gmail.com
EMAIL_HOST_PASSWORD=your-app-password


⚠️  COMMANDES DANGEREUSES (À utiliser avec prudence!)
───────────────────────────────────────────────────────────────────────────

# Supprimer la BD (développement seulement!)
rm db.sqlite3
python manage.py migrate

# Supprimer et recréer tout
python manage.py flush --no-input
python manage.py migrate

# Supprimer une migration
python manage.py migrate apps.experts 0002  # Rollback à cette migration

# Créer une BD à partir de zéro
python manage.py migrate --fake-initial

───────────────────────────────────────────────────────────────────────────
